<script>
let allPapers = [];
let userBookmarks = [];
let paperSummaries = {};

async function loadSummariesData() {
  try {
    const papersResponse = await fetch('papers.json');
    allPapers = await papersResponse.json();
    
    const userResponse = await fetch('http://localhost:5001/api/user/data');
    const userData = await userResponse.json();
    userBookmarks = userData.bookmarks || [];
    
    // Load summaries from user data (if stored)
    const summariesResponse = await fetch('http://localhost:5001/api/user/summaries');
    const summariesData = await summariesResponse.json();
    paperSummaries = summariesData.summaries || {};
    
  } catch (error) {
    console.error('Failed to load data:', error);
  }
}

function generateAISummary(paper) {
  // Simulate AI summary generation
  // In production, this would call an AI API
  const abstract = paper.abstract || '';
  const title = paper.title || '';
  
  // Extract key phrases from title and abstract
  const sentences = abstract.split(/[.!?]+/).filter(s => s.trim().length > 20);
  const keyPoints = sentences.slice(0, 3).map(s => s.trim());
  
  // Generate a concise summary
  const summary = {
    oneLiner: `${title.split(':')[0]}: ${keyPoints[0] || 'A novel approach to AI research'}`,
    keyPoints: keyPoints,
    generatedAt: new Date().toISOString()
  };
  
  return summary;
}

async function saveSummary(arxivId, summary) {
  try {
    await fetch(`http://localhost:5001/api/user/summaries/${arxivId}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ summary })
    });
    paperSummaries[arxivId] = summary;
  } catch (error) {
    console.error('Failed to save summary:', error);
  }
}

function renderSummaries() {
  const container = document.getElementById('summariesList');
  const savedPapers = userBookmarks.map(id => allPapers.find(p => p.arxiv_id === id)).filter(p => p);
  
  document.getElementById('totalPapers').textContent = userBookmarks.length;
  document.getElementById('summaryCount').textContent = Object.keys(paperSummaries).length;
  
  if (savedPapers.length === 0) {
    container.innerHTML = '<div class="no-summary">No papers saved yet. Save papers from the search page to generate summaries.</div>';
    return;
  }
  
  container.innerHTML = savedPapers.map(paper => {
    const summary = paperSummaries[paper.arxiv_id];
    const hasSummary = !!summary;
    
    return `
      <div class="summary-card">
        <div class="summary-header">
          <div class="summary-title">
            <a href="${paper.url}" target="_blank">${paper.title}</a>
            ${hasSummary ? '<span class="summary-status status-generated">✓ Generated</span>' : '<span class="summary-status status-pending">Pending</span>'}
          </div>
          <div class="summary-actions">
            ${!hasSummary ? `<button class="summary-btn" onclick="generateSingleSummary('${paper.arxiv_id}')">🔄 Generate</button>` : ''}
          </div>
        </div>
        <div class="summary-content">
          ${hasSummary ? `
            <p><strong>Quick Summary:</strong> ${summary.oneLiner}</p>
            ${summary.keyPoints && summary.keyPoints.length > 0 ? `
              <p><strong>Key Points:</strong></p>
              <ul>
                ${summary.keyPoints.map(point => `<li>${point}</li>`).join('')}
              </ul>
            ` : ''}
          ` : `
            <p class="loading">Click "Generate" to create an AI summary for this paper.</p>
          `}
        </div>
        <div class="summary-meta">
          <div class="summary-meta-item">📅 ${paper.date}</div>
          <div class="summary-meta-item">👥 ${paper.authors}</div>
          <div class="summary-meta-item">🏷️ ${paper.topics ? paper.topics.join(', ') : 'N/A'}</div>
        </div>
      </div>
    `;
  }).join('');
}

async function generateSingleSummary(arxivId) {
  const paper = allPapers.find(p => p.arxiv_id === arxivId);
  if (!paper) return;
  
  const summary = generateAISummary(paper);
  await saveSummary(arxivId, summary);
  renderSummaries();
}

async function generateAllSummaries() {
  const savedPapers = userBookmarks.map(id => allPapers.find(p => p.arxiv_id === id)).filter(p => p);
  const papersWithoutSummaries = savedPapers.filter(p => !paperSummaries[p.arxiv_id]);
  
  if (papersWithoutSummaries.length === 0) {
    alert('All papers already have summaries!');
    return;
  }
  
  const btn = document.getElementById('generateAllBtn');
  btn.textContent = '⏳ Generating...';
  btn.disabled = true;
  
  for (const paper of papersWithoutSummaries) {
    const summary = generateAISummary(paper);
    await saveSummary(paper.arxiv_id, summary);
    // Small delay to avoid overwhelming the API
    await new Promise(resolve => setTimeout(resolve, 100));
  }
  
  btn.textContent = '🔄 Generate All Summaries';
  btn.disabled = false;
  renderSummaries();
}

async function clearAllSummaries() {
  if (!confirm('Are you sure you want to clear all summaries? This cannot be undone.')) {
    return;
  }
  
  try {
    for (const arxivId of Object.keys(paperSummaries)) {
      await fetch(`http://localhost:5001/api/user/summaries/${arxivId}`, {
        method: 'DELETE'
      });
    }
    paperSummaries = {};
    renderSummaries();
  } catch (error) {
    console.error('Failed to clear summaries:', error);
  }
}

// Event listeners
document.getElementById('generateAllBtn').addEventListener('click', generateAllSummaries);
document.getElementById('clearAllBtn').addEventListener('click', clearAllSummaries);

// Initialize
document.addEventListener('DOMContentLoaded', () => {
  loadSummariesData().then(() => {
    renderSummaries();
  });
});
</script>
