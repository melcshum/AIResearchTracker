// Count user contributions for this topic
function countContributions() {
  let count = 0;
  for (let i = 0; i < localStorage.length; i++) {
    const key = localStorage.key(i);
    if (key.startsWith('wiki_contributions_') || key.startsWith('concept_validation_') || key.startsWith('feedback_')) {
      const data = JSON.parse(localStorage.getItem(key) || '[]');
      if (Array.isArray(data)) count += data.length;
      else count += 1;
    }
  }
  const el = document.getElementById('hitlContribCount');
  if (el) el.textContent = count;
}

countContributions();
